bash <<'NEBULA_EOF' 2>&1 | tee ~/nebula-diag6.txt
# =============================================================================
#  NEBULA DIAGNOSTICS round 7 — COPY AND PASTE THIS FIRST,
#  THEN CLICK "Launch & Inject" while it samples.
#
#  The one question this answers: does the hook now SURVIVE Sober's re-exec?
#
#  Established in v0.4.36:
#    * Sober re-execs itself ("required by sober" then "required by
#      /proc/self/exe"). The 2nd exec scrubs the environment, so LD_PRELOAD
#      is gone -> the hook vanished at ~8s. That is why hook=no everywhere.
#    * Fix: the hook writes ITSELF into /etc/ld.so.preload, which ld.so reads
#      off disk and which environment scrubbing cannot touch.
#    * ptrace is now OFF by default. It killed Sober, it never worked.
#      (There is an "Allow ptrace inject (unsafe)" toggle; leave it OFF.)
# =============================================================================

sec() { printf '\n\n========== %s ==========\n' "$1"; }

sec "SAMPLING 45s — CLICK 'Launch & Inject' NOW"
printf '%-7s %-8s %-8s %-6s %-6s %-6s %-6s\n' time pid comm hook badcpu loader preloads
for i in $(seq 1 90); do
  T=$(i/2)
  for pid in $(pgrep -f 'sober|org\.vinegarhq\.Sober' 2>/dev/null); do
    [ -r "/proc/$pid/maps" ] || continue
    COMM=$(cat /proc/$pid/comm 2>/dev/null)
    HK=$(grep -c libnebula_hook /proc/$pid/maps 2>/dev/null || echo 0)
    BC=$(grep -c libbadcpu      /proc/$pid/maps 2>/dev/null || echo 0)
    LD=$(grep -c libloader     /proc/$pid/maps 2>/dev/null || echo 0)
    # Did the hook manage to arm the file-based preload inside the sandbox?
    PL=$(cat /proc/$pid/root/etc/ld.so.preload 2>/dev/null | grep -c nebula_hook || echo 0)
    printf '%-7s %-8s %-8s %-6s %-6s %-6s %-6s\n' "${T}s" "$pid" "$COMM" "$HK" "$BC" "$LD" "$PL"
  done
  sleep 0.5
done

sec "WHAT THE HOOK ARMED (per process root)"
for pid in $(pgrep -f 'sober|org\.vinegarhq\.Sober' 2>/dev/null); do
  echo "--- pid $pid ($(cat /proc/$pid/comm 2>/dev/null)) ---"
  cat /proc/$pid/root/etc/ld.so.preload 2>&1 | sed 's/^/    /'
done

sec "HOOK LOG (~/.nebula/queue/log) — now via NEBULA_QUEUE_DIR"
echo "mtime: $(stat -c '%y' ~/.nebula/queue/log 2>&1)"
tail -n 60 ~/.nebula/queue/log 2>&1

sec "HOOK STATUS"
echo "status: $(cat ~/.nebula/queue/status 2>&1)"

sec "EVERY MAPPED .so IN THE MAIN SOBER PROCESS"
MAIN=$(pgrep -f 'org\.vinegarhq\.Sober' | tail -1)
if [ -n "$MAIN" ]; then
  echo "pid $MAIN:"
  grep -oE '/[^ ]*\.so[^ ]*' /proc/$MAIN/maps 2>/dev/null | sort -u | sed 's/^/    /'
else
  echo "(sober not running)"
fi

sec "DONE — send me ~/nebula-diag6.txt"
# =============================================================================
NEBULA_EOF
